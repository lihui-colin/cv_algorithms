cmake_minimum_required(VERSION 3.16)
project(shape_match VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SHAPE_MATCH_NATIVE "Optimize for build machine (not portable)" OFF)
find_package(Threads REQUIRED)
add_library(shape_match src/types.cpp src/model.cpp src/geometry.cpp src/search.cpp src/result.cpp src/parallel.cpp src/precision.cpp)
target_include_directories(shape_match PUBLIC include PRIVATE src)
target_link_libraries(shape_match PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(shape_match PRIVATE /W4)
else()
  target_compile_options(shape_match PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SHAPE_MATCH_NATIVE AND NOT MSVC)
  target_compile_options(shape_match PRIVATE -O3 -march=native)
endif()
add_executable(match_sample examples/match_sample.cpp)
target_link_libraries(match_sample PRIVATE shape_match)
add_executable(test_match tests/test_match.cpp)
target_link_libraries(test_match PRIVATE shape_match)
add_executable(accuracy_bench tests/accuracy_bench.cpp)
target_link_libraries(accuracy_bench PRIVATE shape_match)
add_executable(performance_bench tests/performance_bench.cpp)
target_link_libraries(performance_bench PRIVATE shape_match)
add_executable(refinement_probe tests/refinement_probe.cpp)
target_link_libraries(refinement_probe PRIVATE shape_match)
add_executable(halcon_parameter_probe tests/halcon_parameter_probe.cpp)
target_link_libraries(halcon_parameter_probe PRIVATE shape_match)
add_executable(refinement_trajectory tests/refinement_trajectory.cpp)
target_link_libraries(refinement_trajectory PRIVATE shape_match)
enable_testing()
add_executable(test_scoring tests/test_scoring.cpp)
target_link_libraries(test_scoring PRIVATE shape_match)
add_test(NAME scoring_equivalence COMMAND test_scoring)
add_executable(test_normal_profile tests/test_normal_profile.cpp)
target_link_libraries(test_normal_profile PRIVATE shape_match)
add_test(NAME normal_profile_observations COMMAND test_normal_profile)
add_executable(test_precision tests/test_precision.cpp)
target_link_libraries(test_precision PRIVATE shape_match)
add_test(NAME continuous_precision COMMAND test_precision)
set_tests_properties(continuous_precision PROPERTIES ENVIRONMENT "SHAPE_MATCH_NUM_THREADS=4" TIMEOUT 60)
foreach(method nearest_point gradient_gaussian)
  add_test(NAME accuracy_${method} COMMAND accuracy_bench
    ${CMAKE_CURRENT_BINARY_DIR}/accuracy_${method}.csv ${method} 1.5 extended)
  set_tests_properties(accuracy_${method} PROPERTIES
    ENVIRONMENT "SHAPE_MATCH_NUM_THREADS=4" TIMEOUT 180)
endforeach()
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME refinement_trajectory_equivalence COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_refinement_trajectory.py
    $<TARGET_FILE:match_sample> $<TARGET_FILE:refinement_trajectory> ${CMAKE_CURRENT_SOURCE_DIR}/data)
  set_tests_properties(refinement_trajectory_equivalence PROPERTIES
    ENVIRONMENT "SHAPE_MATCH_NUM_THREADS=4" TIMEOUT 60)
  add_test(NAME gt_comparison COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_compare_gt.py)
endif()
add_test(NAME contract_and_geometry COMMAND test_match)
add_executable(test_parallel tests/test_parallel.cpp)
target_link_libraries(test_parallel PRIVATE shape_match)
foreach(threads 1 4 invalid)
  add_test(NAME parallel_${threads} COMMAND test_parallel)
  set_tests_properties(parallel_${threads} PROPERTIES
    ENVIRONMENT "SHAPE_MATCH_NUM_THREADS=${threads}" TIMEOUT 30)
endforeach()
install(TARGETS shape_match match_sample)
install(DIRECTORY include/ DESTINATION include)
