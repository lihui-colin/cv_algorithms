cmake_minimum_required(VERSION 3.16)
project(shape_match VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SHAPE_MATCH_NATIVE "Optimize for build machine (not portable)" OFF)
find_package(Threads REQUIRED)
add_library(shape_match src/types.cpp src/model.cpp src/geometry.cpp src/search.cpp src/result.cpp)
target_include_directories(shape_match PUBLIC include PRIVATE src)
target_link_libraries(shape_match PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(shape_match PRIVATE /W4)
else()
  target_compile_options(shape_match PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SHAPE_MATCH_NATIVE AND NOT MSVC)
  target_compile_options(shape_match PRIVATE -O3 -march=native)
endif()
add_executable(match_sample examples/match_sample.cpp)
target_link_libraries(match_sample PRIVATE shape_match)
add_executable(test_match tests/test_match.cpp)
target_link_libraries(test_match PRIVATE shape_match)
add_executable(accuracy_bench tests/accuracy_bench.cpp)
target_link_libraries(accuracy_bench PRIVATE shape_match)
add_executable(performance_bench tests/performance_bench.cpp)
target_link_libraries(performance_bench PRIVATE shape_match)
enable_testing()
add_test(NAME contract_and_geometry COMMAND test_match)
install(TARGETS shape_match match_sample)
install(DIRECTORY include/ DESTINATION include)
